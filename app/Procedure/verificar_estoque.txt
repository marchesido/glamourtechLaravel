DELIMITER $$

CREATE FUNCTION verificar_estoque(produtoId INT, qtd INT)
RETURNS TINYINT
DETERMINISTIC
BEGIN
    DECLARE estoque_atual INT;

    SELECT estoque INTO estoque_atual
    FROM produtos
    WHERE id = produtoId;

    IF estoque_atual IS NULL THEN
        RETURN 0; -- produto não existe
    END IF;

    IF estoque_atual < qtd THEN
        RETURN 0; -- sem estoque suficiente
    END IF;

    RETURN 1; -- estoque OK
END$$

DELIMITER ;
